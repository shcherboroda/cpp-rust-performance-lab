function(add_mixed_lifecycle target library version_define)
    add_executable(${target} benchmarks/mixed_lifecycle.cpp)
    target_compile_definitions(${target} PRIVATE ${version_define})
    target_link_libraries(${target} PRIVATE ${library})
endfunction()

function(add_mixed_lifecycle_with_bbo target library version_define)
    add_mixed_lifecycle(${target} ${library} ${version_define})
    target_compile_definitions(${target} PRIVATE LLAB_READ_BBO)
endfunction()
